function handleGetDailyReport(p) { const auth = requireAuth(p); if (!auth) return { success:false, message:'Session หมดอายุ' }; const dateStr = p.date; const s = getSettingsMap(); const year = String(s.academic_year || '').trim(); if (!year) return { success:true, date: dateStr, records: [] }; const cache = CacheService.getScriptCache(); const cacheKey = `daily_${year}_${dateStr}`; const cached = cache.get(cacheKey); if (cached) { return { success:true, date: dateStr, records: JSON.parse(cached) }; } const uSheet = getOrCreateSpreadsheet().getSheetByName(SHEET_USERS); const uData = uSheet.getDataRange().getValues(); const allowedUsers = {}; for (let i = 1; i < uData.length; i++) { if (uData[i][4] === 'approved' && uData[i][3] !== 'admin') { allowedUsers[String(uData[i][2]).trim()] = { profilePic: uData[i][13] || '' // 🟢 ดึงรูปโปรไฟล์ }; } } const records = []; ['1', '2'].forEach(t => { const aSheet = getAttendanceSheet(year, t, false); if (!aSheet) return; const data = aSheet.getDataRange().getValues(); for (let i = 1; i < data.length; i++) { const rowDateStr = getFormattedDate(data[i][0]); const rowName = String(data[i][1]).trim(); if (rowDateStr === dateStr && allowedUsers[rowName]) { if (!records.find(r => r.fullName === rowName)) { records.push({ fullName: rowName, profilePic: allowedUsers[rowName].profilePic, // 🟢 ส่งรูปโปรไฟล์กลับไป inTime: formatTime(data[i][2]), outTime: formatTime(data[i][3]), note: data[i][4] || '-', rawInTime: cleanStr(data[i][2]) || '99:99' }); } } } }); records.sort((a, b) => a.rawInTime.localeCompare(b.rawInTime)); cache.put(cacheKey, JSON.stringify(records), 30); return { success:true, date: dateStr, records }; } function handleGetReports(p) { const auth = requireAuth(p); if (!auth) return { success:false, message:'Session หมดอายุ' }; const targetYear = String(p.year || '').trim(); const targetTerm = String(p.term || '').trim(); const targetMonth = String(p.month || 'all'); if (!targetYear || (targetTerm !== '1' && targetTerm !== '2')) { return { success:true, records: [] }; } const cache = CacheService.getScriptCache(); const cacheKey = `report_${targetYear}_${targetTerm}_${targetMonth}_${auth.username}`; const cached = cache.get(cacheKey); if (cached) { return { success:true, records: JSON.parse(cached) }; } const aSheet = getAttendanceSheet(targetYear, targetTerm, false); if (!aSheet) return { success:true, records: [] }; const data = aSheet.getDataRange().getValues(); const records = []; const uSheet = getOrCreateSpreadsheet().getSheetByName(SHEET_USERS); const uData = uSheet.getDataRange().getValues(); // สร้าง Map ชื่อ -> รูปโปรไฟล์ const userPicMap = {}; for (let i = 1; i < uData.length; i++) { userPicMap[String(uData[i][2]).trim()] = uData[i][13] || ''; } const uRow = findUserRow(uSheet, auth.username); const uRowData = uSheet.getRange(uRow, 1, 1, 17).getValues()[0]; const userPos = String(uRowData[14] || '').trim(); for (let i = 1; i < data.length; i++) { const rowName = String(data[i][1]).trim(); if (auth.role !== 'admin' && userPos !== 'ผอ.' && rowName !== auth.fullName.trim()) continue; const dateStr = getFormattedDate(data[i][0]); const dParts = dateStr.split('/'); if (dParts.length !== 3) continue; if (targetMonth !== 'all' && dParts[1] !== targetMonth) continue; records.push({ date: dateStr, name: rowName, profilePic: userPicMap[rowName] || '', // 🟢 ส่งรูปโปรไฟล์กลับไป inTime: formatTime(data[i][2]), outTime: formatTime(data[i][3]), note: data[i][4] || '-', rawInTime: cleanStr(data[i][2]) || '99:99' }); } records.sort((a, b) => { const [dA, mA, yA] = a.date.split('/').map(Number); const [dB, mB, yB] = b.date.split('/').map(Number); if (yA !== yB) return yA - yB; if (mA !== mB) return mA - mB; if (dA !== dB) return dA - dB; return a.rawInTime.localeCompare(b.rawInTime); }); cache.put(cacheKey, JSON.stringify(records), 30); return { success:true, records }; }
